FOR FILE COMPRESSION:
---------------------
# Compress entire folder
await file("compress", src="C:/MyProject", dst="project.zip")

# Extract archive
await file("extract", src="project.zip", dst="C:/Restored")

# List archive contents
await file("list_archive", src="project.zip")

# Compress specific files
await file("compress_files", 
           name="doc1.pdf,doc2.xlsx,readme.txt", 
           dst="important_docs.zip")

# TAR.GZ format (auto-detected from extension)
await file("compress", src="C:/Data", dst="backup.tar.gz")
